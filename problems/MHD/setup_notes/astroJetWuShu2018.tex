\section{DynSGS as it runs for this case}
\cm{$\Delta_K=\Delta^{elem}_K/(N+1)$} $=0.00125$, $h=\Delta t$.
\begin{enumerate}[leftmargin=*]
\item \textbf{History, no hold.} \cm{Rotation once per step}; with $H=0$ the residual is live from the first RHS
call, on a history seeded with the initial condition (first stage of step~1: BDF2 of $(q^0,q^0,q^0)$, $D_tq=0$).
\item \textbf{Element residual} (residual sensor, stage-consistent stencil):
$R^s_{K,I}=|D_tq^s_I-r^{K,s}_I/m^K_I|$, $D_tq$ = BDF2 on $(q^n,q^{n-1},q^{n-2})$ at $\tau=0$, the three-point
derivative through $(q(\tau),q^n,q^{n-1})$ at $\tau>0$; $r^K$ the element's own weak inviscid RHS.
\item \textbf{Dirichlet nodes.} Every node on a non-periodic boundary edge (nozzle and the rest of the bottom,
axis, right, top) is listed by \code{\_dsgs\_boundary\_pairs!}, once per element containing it; its entries are replaced by $m^K_ID_tq_I$, so
$R\equiv0$ there in all nine slots. Inflow shocks are sensed by the interior nodes of the same elements.
\item \textbf{Element normalization with fast floors} over the 25 nodes of $K$:
$n^s_K=\max(S^s_K,\sigma^s_K)+10^{-16}$, $\sigma_K=(\bar\rho_K,\ \bar\rho_K\bar c_{f,K},\ \gamma\bar p_K+|\bar\bB_K|^2,\ \sqrt{\gamma\bar p_K+|\bar\bB_K|^2})$
for $(\rho,\bmom,E,\bB)$, $\bar c_{f,K}=\sqrt{(\gamma\bar p_K+|\bar\bB_K|^2)/\bar\rho_K}$.
\item \textbf{Ratio, cap, coefficient.} \cm{$\mathcal R_K=\max_{I\in K}\max_{s\le8}R^s_{K,I}/n^s_K$},
\cm{$w_K=\max_I(|\bv_I|+\sqrt{\max(\gamma p_I/\rho_I+|\bB_I|^2/\rho_I,0)})$},
$\nu_K=\max(0,\min(0.5\,\Delta_Kw_K,\ \Delta_K^2\mathcal R_K))$, no floor, no cutoff, \cm{one value per element},
\cm{recomputed at every stage}; $\nu_K$ on all nine slots.
\end{enumerate}

\section{The viscous operator}
\code{user\_primitives.jl} returns $\bq$ itself:
$\partial_t\bq+\nabla\cdot\bF=\nabla\cdot(\nu_K\nabla\bq)$ on $(\rho,\rho u,\rho v,E,\rho w,B_x,B_y,B_z,\psi)$ ---
\mj{a Laplacian of the conserved state}, \cm{mass diffusion on}, \mj{no $\tau\cdot\bv$ or resistive-work term}.

\subsection*{Flux-only guards (\code{user\_flux.jl:51--58})}
Inside \code{user\_flux!} only: $\rho_g=\max(\rho,10^{-14})$ in the velocities and in the kinetic energy of
\code{pressure\_mhd}, and $p\leftarrow\max(p,10^{-6})$ (\code{p\_floor\_mhd}, equal to $\eps_p$). They change
the flux of a node that is momentarily inadmissible, not the state; with the limiter active every node the flux
sees already satisfies $\rho\ge\eps_\rho$, $p\ge\eps_p$ up to the round-off margins above.

\section{Observations}
\begin{enumerate}[leftmargin=*]
\item Deck and comments agree with the code. The \code{"residual"} sensor is chosen explicitly for its
boundary-node treatment, which the \code{"legacy"} path does not have.
\item $p$ in this beam is $\approx5.6\times10^{-6}$ of $E$; the deck comment on \code{:dsgs\_norms} explains the
element norms by that ratio, and \code{Positivity.jl:275--289} quantifies how small $p_{\min}$ may usefully be
($(\gamma-1)\,\mathrm{ulp}(E)\approx2.3\times10^{-11}$ against $p_{\min}=10^{-6}$).
\item \textbf{[Changed in \code{b956ccd}.]} With the residual sensor this deck now runs with the coefficient frozen
per step, the \code{DSGS\_MHD} default (\code{:dsgs\_freeze\_stage}; see the Brio--Wu note for the check). Not re-run here.
\end{enumerate}
