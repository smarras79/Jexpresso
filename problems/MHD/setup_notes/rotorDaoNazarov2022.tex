\section{DynSGS as it runs for this case}
$\Delta^{elem}_K=1/128$, \cm{$\Delta_K=\Delta^{elem}_K/(N+1)$} $=1/512$, $h=\Delta t$.
\begin{enumerate}[leftmargin=*]
\item \cm{Rotation once per step}; \mj{two held steps} ($\nu=0$).
\item \textbf{Legacy residual}, every stage: $R^s_I=\big|\tfrac{3q^s_I-4q^{n,s}_I+q^{n-1,s}_I}{2h}-(M^{-1}\mathcal R_{inv})^s_I\big|$,
$\mathcal R_{inv}$ assembled over ranks and periodic copies; the state $q$ is the already-limited stage state.
\item \textbf{Element normalization}: $n^s_K=\max(S^s_K,\sigma^s_K)+10^{-16}$ over the 16 nodes of $K$, with
$\sigma_K=(\bar\rho_K,\ \bar\rho_K\bar c_K,\ \bar\rho_K\bar c_K^2,\ \sqrt{\bar\rho_K}\bar c_K)$ for $(\rho,\ \bmom,\ E,\ \bB)$,
$\bar c_K=\sqrt{\gamma\bar p_K/\bar\rho_K}$ (sound speed).
\item \cm{$\mathcal R_K=\max_{I\in K}\max_{s\le8}R^s_I/n^s_K$}, \cm{$w_K=\max_I(|\bv_I|+\sqrt{\max(\gamma p_I/\rho_I+|\bB_I|^2/\rho_I,0)})$},
$\nu_K=\max(0,\min(0.5\,\Delta_Kw_K,\ \Delta_K^2\mathcal R_K))$; no floor, no cutoff; \cm{one value per element},
\cm{recomputed at every stage}; $\nu_K$ on all nine slots.
\end{enumerate}

\section{The viscous operator}
\code{user\_primitives.jl} copies $\bq$ (the energy split is coded but inactive, \code{:dsgs\_nazarov\_energy => false}), so
\[
\partial_t\bq+\nabla\cdot\bF(\bq)=\nabla\cdot(\nu_K\nabla\bq)\qquad\text{on }(\rho,\rho u,\rho v,E,\rho w,B_x,B_y,B_z,\psi),
\]
\mj{a Laplacian of the conserved state}, \cm{mass diffusion on}, \mj{no $\tau\cdot\bv$ or resistive-work term}.
