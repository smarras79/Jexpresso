\section{DynSGS as it runs for this case}
$\Delta^{elem}_K=1/64$, \cm{$\Delta_K=\Delta^{elem}_K/(N+1)$} $=1/256$, $h=\Delta t$.
\begin{enumerate}[leftmargin=*]
\item \textbf{History, no hold.} \cm{Rotation once per step}. With $H=0$ the hold is off (\code{rhs.jl:183}),
so on step~1 the history is the initial condition: the legacy weights act on $(q,q^n,q^{n-1})=(q^0,q^0,q^0)$
at the first stage, $D_tq=0$, and $R=|M^{-1}\mathcal R_{inv}|$ is the full flux divergence of the initial jump.
\item \textbf{Legacy residual}, every stage: $R^s_I=\big|\tfrac{3q^s_I-4q^{n,s}_I+q^{n-1,s}_I}{2h}-(M^{-1}\mathcal R_{inv})^s_I\big|$,
assembled RHS; signed values $\hat R^s_I$ (without the modulus) feed the thermal residual below.
\item \textbf{Element normalization with fast floors}: $n^s_K=\max(S^s_K,\sigma^s_K)+10^{-16}$ over the 16 nodes of
$K$, with $\bar c_{f,K}=\sqrt{(\gamma\bar p_K+|\bar\bB_K|^2)/\bar\rho_K}$ in place of the sound speed:
$\sigma_K=(\bar\rho_K,\ \bar\rho_K\bar c_{f,K},\ \gamma\bar p_K+|\bar\bB_K|^2,\ \sqrt{\gamma\bar p_K+|\bar\bB_K|^2})$
for $(\rho,\ \bmom,\ E,\ \bB)$.
\item \textbf{Thermal-energy residual} (\code{SGS.jl:1770--1788, 1814--1825}). At every node,
\[
\hat R_{\rho e}=\hat R_E-\bv\cdot\hat R_{\bmom}+\tfrac12|\bv|^2\hat R_\rho-\bB\cdot\hat R_{\bB}-\psi\hat R_\psi,
\]
normalized by $n^{\rho e}_K=\max\big(\max(e_{\max}-\bar e,\ \bar e-e_{\min}),\ r_{loc}\gamma\bar p_K\big)+10^{-16}$,
$e=E-\tfrac12|\bmom|^2/\rho-\tfrac12|\bB|^2-\tfrac12\psi^2$ at the nodes of $K$, $\bar e$ their plain average.
$|\hat R_{\rho e}|/n^{\rho e}_K$ joins the max.
\item \textbf{Ratio, cap, coefficient.}
\cm{$\mathcal R_K=\max_{I\in K}\max\big(\max_{s\le8}R^s_I/n^s_K,\ |\hat R_{\rho e,I}|/n^{\rho e}_K\big)$}
(underlined part: the eight conserved slots),
\cm{$w_K=\max_I(|\bv_I|+\sqrt{\max(\gamma p_I/\rho_I+|\bB_I|^2/\rho_I,0)})$},
$\nu_K=\max(0,\min(0.5\,\Delta_Kw_K,\ \Delta_K^2\mathcal R_K))$, no floor, no cutoff,
\cm{one value per element}, \cm{recomputed at every stage}; $\nu_K$ on all nine slots.
\end{enumerate}

\section{The viscous operator}
\code{user\_primitives.jl} copies $\bq$ (energy split coded, inactive):
$\partial_t\bq+\nabla\cdot\bF=\nabla\cdot(\nu_K\nabla\bq)$ on all nine slots --- \mj{a Laplacian of the conserved
state}, \cm{mass diffusion on}, \mj{no $\tau\cdot\bv$ or resistive-work term}.

the limited state.
