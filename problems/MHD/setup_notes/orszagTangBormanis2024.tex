\section{DynSGS as it runs for this case}
$\Delta^{elem}_K=1/64$, \cm{$\Delta_K=\Delta^{elem}_K/(N+1)$} $=1/320$; $h=\Delta t$.
\begin{enumerate}[leftmargin=*]
\item \textbf{History and hold.} \cm{Rotation once per step} (\code{rhs.jl:936}); \mj{two held steps}, during
which $\nu=0$ ($\Cmin=0$).
\item \textbf{Legacy residual} (\code{rhs.jl:960--964}, \code{\_dsgs\_residual\_rhs!}):
at \emph{every} stage, with the stage value $q_I$,
\[
R^s_I=\Big|\frac{3q^s_I-4q^{n,s}_I+q^{n-1,s}_I}{2h}-\big(M^{-1}\mathcal R_{inv}\big)^s_I\Big|,
\]
$\mathcal R_{inv}$ the inviscid right-hand side (flux + $\psi$ source) assembled over ranks and periodic
copies (\code{DSS\_global\_RHS!} on a copy), $M$ the lumped global mass. The value is written into the
element layout so the kernel sees the same number from every element sharing $I$. No Dirichlet zeroing
on this path (and none needed: periodic).
\item \textbf{Normalization} (domain): $n^s=\max(S^s,\sigma^s)+10^{-16}$ with the mean-state floors
$\sigma^\rho=\bar\rho$, $\sigma^{\bmom}=\bar\rho\bar c$, $\sigma^E=\bar\rho\bar c^2$, $\sigma^{\bB}=\sqrt{\bar\rho}\,\bar c$, sound speed
$\bar c=\sqrt{\gamma\bar p/\bar\rho}$. $\rho$ is uniform at $t=0$, so its residual is first measured against $\bar\rho$.
\item \textbf{Ratio, cap, coefficient.} \cm{$\mathcal R_K=\max_{I\in K}\max_{s\le8}R^s_I/n^s$};
\cm{$w_K=\max_I(|\bv_I|+\sqrt{\max(\gamma p_I/\rho_I+|\bB_I|^2/\rho_I,0)})$};
$\nu_K=\max(0,\min(0.5\,\Delta_Kw_K,\ \Delta_K^2\mathcal R_K))$, no floor,
\cm{one value per element}, \cm{recomputed at every stage}.
\item \textbf{Slot coefficients} (physical branch, \code{SGS.jl:1889--1900}), with $\bar\rho_K$ the plain
average of nodal $\rho$ over the $(N+1)^2$ nodes of $K$:
\[
\mu^\rho=\nu_K,\quad \mu^{\rho u}=\mu^{\rho v}=\mu^{\rho w}=\bar\rho_K\nu_K,\quad
\mu^{E}=\bar\rho_K\nu_K/\Pr=\bar\rho_K\nu_K,\quad \mu^{B}=\nu_K,\quad\mu^\psi=\nu_K .
\]
\end{enumerate}

\code{\_expansion\_visc!} (\code{rhs.jl:2943}) assembles:

\toprule

$\rho w$ & $\bar\rho\nu\nabla w$ & $\bar\rho\nu$\\
$E$ & $\bar\rho\nu\nabla T+\boldsymbol\tau\cdot\bv+\nu\sum_kB_k\nabla B_k$ & heat $+$ viscous work $+$ resistive work\\
$B_k$ & $\nu\nabla B_k$ & $\eta=\nu$\\
$\psi$ & $\nu\nabla\psi$ & \\

\end{center}
The work terms (\code{rhs.jl:3076--3122}) return the in-plane kinetic and the magnetic energy removed by the
stress and the induction Laplacian to $E$. The $\rho w$ and $\psi$ dissipations have no matching work term
($w\equiv0$ in this problem).

\section{Positivity}
\begin{itemize}[leftmargin=*]
\item \mj{\code{:lpositivity} false}, \mj{no stage limiter, no state modification}.
\item No flux-only floor: \code{pressure\_mhd} of this case is unguarded, so a negative $p$ reaches the
flux as is.
\item Admissibility therefore rests entirely on the \dsgs{} dissipation; the deck records why the
Boyd--Vandeven filter is kept off (it filters $\rho$, $\rho\bv$, $E$ independently and can drive $p<0$).
\end{itemize}

\section{Observations: code versus comments}
\begin{enumerate}[leftmargin=*]
\item The comment above \code{:dsgs\_norms} says ``the code-wide default is \code{"rank"}''; the default in
\code{mod\_inputs.jl:1161} is \code{"domain"}, which is also what the deck sets.

$128^2$ of the paper''; the deck sets $\Delta t=2\times10^{-4}$ and refines the mesh once
($64\times64$ elements, $256^2$ points).
\item \code{:output\_dir} points to a cluster scratch path (\code{/scratch/smarras/\dots}).
\end{enumerate}
